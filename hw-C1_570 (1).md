## Question 3

#include <stdio.h>
#include <math.h>

double calculate_plus_root(double a, double b, double c) {
    double discriminant = b * b - 4 * a * c;
    return (-b + sqrt(discriminant)) / (2 * a);
}
double calculate_minus_root(double a, double b, double c) {
    double discriminant = b * b - 4 * a * c;
    return (-b - sqrt(discriminant)) / (2 * a);
}

int main() {
    double a = 1.0;
    double b = 2.0;
    double c = -4.0;
    double root1 = calculate_plus_root(a, b, c);
    double root2 = calculate_minus_root(a, b, c);
    
    printf("Equation: (%.0f)x^2 + (%.0f)x + (%.0f) = 0\n\n", a, b, c);
    printf("Root 1 (b+ part): %.4f\n", root1);
    printf("Root 2 (b- part): %.4f\n", root2);

}
