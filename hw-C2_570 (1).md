            matrix[i][j] = i * 10 + j;
        }
    }
    printf("=== Original Matrix ===\n");
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            printf("%2d ", matrix[i][j]);
        }
        printf("\n");
    }
    printf("\n");
    for (int i = 0; i < 4; i++) {
        for (int j = i + 1; j < 4; j++) {
            int temp = matrix[i][j];
            matrix[i][j] = matrix[j][i];
            matrix[j][i] = temp;
        }
    }
    printf("=== Transposed Matrix ===\n");
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            printf("%2d ", matrix[i][j]);
        }
        printf("\n");
    }
    printf("\n");
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            rowsum[i] += matrix[i][j];
            colsum[j] += matrix[i][j];
        }
    }
    printf("=== Row Sums ===\n");
    for (int i = 0; i < 4; i++) {
        printf("rowsum[%d] = %d\n", i, rowsum[i]);
    }
    printf("\n");
    printf("=== Column Sums ===\n");
    for (int j = 0; j < 4; j++) {
        printf("colsum[%d] = %d\n", j, colsum[j]);
