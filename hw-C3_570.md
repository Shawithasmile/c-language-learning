#include <stdio.h>
#include <stdlib.h>
char calculate_grade(double score) {
    if (score >= 90) return 'A';
    if (score >= 80) return 'B';
    if (score >= 70) return 'C';
    if (score >= 60) return 'D';
    return 'F';
}
int main() {
    int n;
    printf("Enter number of courses: ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        printf("Invalid number of courses.\n");
        return 1;
    }
    char (*course)[6] = malloc(n * sizeof(*course));
    double *scores = malloc(n * sizeof(double));
    if (course == NULL || scores == NULL) {
        printf("Memory allocation failed.\n");
        free(course);
        free(scores);
        return 1;
    }
    for (int i = 0; i < n; i++) {
        printf("\n--- Course %d ---\n", i + 1);
        printf("Enter Course ID (e.g., cp121): ");
        scanf("%5s", course[i]);
        
        printf("Enter Score (0-100): ");
        scanf("%lf", &scores[i]);
    }
    printf("\n%-15s %-10s %-5s\n", "Course ID", "Score", "Grade");
    printf("-----------------------------------\n");
    for (int i = 0; i < n; i++) {
        char grade = calculate_grade(scores[i]);
        printf("%-15s %-10.2f %-5c\n", course[i], scores[i], grade);
    }
    free(course);
    free(scores);
    return 0;
}

#include <stdio.h>

int main() {
    const char *day[] = {"Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"};
    int n = sizeof(day) / sizeof(day[0]);
    printf("--- all days (indexing) ---\n");
    for (int i = 0; i < n; i++) {
        printf("day[%d] = %s\n", i, day[i]);
    }
    printf("\n--- all days (pointer) ---\n");
    for (int i = 0; i < n; i++) {
        printf("*(day+%d) = %s\n", i, *(day + i));
    }
    printf("\n--- lengths ---\n");
    for (int i = 0; i < n; i++) {
        const char *p = *(day + i);
        int len = 0;
        while (*p != '\0') {
            len++;
            p++;
        }
        printf("%s: %d\n", *(day + i), len);
    }
    printf("\nDays starting with 'W': ");
    for (int i = 0; i < n; i++) {
        if (**(day + i) == 'W') {
            printf("%s ", *(day + i));
        }
    }
    printf("\n");
    return 0;
}
